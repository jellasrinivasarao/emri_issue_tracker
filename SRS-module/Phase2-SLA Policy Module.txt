CREATE TABLE cfg_sla_policy
(
    sla_policy_id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    project_id BIGINT UNSIGNED NOT NULL,

    application_id BIGINT UNSIGNED NOT NULL,

    service_id BIGINT UNSIGNED NOT NULL,

    priority_id BIGINT UNSIGNED NOT NULL,

    calendar_id BIGINT UNSIGNED NOT NULL,

    response_time_minutes INT UNSIGNED NOT NULL,

    resolution_time_minutes INT UNSIGNED NOT NULL,

    warning_before_minutes INT UNSIGNED DEFAULT 0,

    auto_escalation TINYINT(1) DEFAULT 1,

    description TEXT NULL,

    is_active TINYINT(1) DEFAULT 1,

    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

    updated_at DATETIME NULL,

    CONSTRAINT fk_sla_project
        FOREIGN KEY(project_id)
        REFERENCES mst_project(project_id),

    CONSTRAINT fk_sla_application
        FOREIGN KEY(application_id)
        REFERENCES mst_application(application_id),

    CONSTRAINT fk_sla_service
        FOREIGN KEY(service_id)
        REFERENCES mst_service(service_id),

    CONSTRAINT fk_sla_priority
        FOREIGN KEY(priority_id)
        REFERENCES mst_priority(priority_id),

    CONSTRAINT fk_sla_calendar
        FOREIGN KEY(calendar_id)
        REFERENCES mst_working_calendar(calendar_id)
);
